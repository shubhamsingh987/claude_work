[global]
type=global
endpoint_identifier_order=ip,username,anonymous
[transport-udp]
type=transport
protocol=udp
bind=0.0.0.0
[pbx_ata]
type=endpoint
context=from-ata
disallow=all
allow=ulaw
allow=alaw
aors=pbx_ata-aor
direct_media=no
[pbx_ata-aor]
type=aor
contact=sip:192.168.1.24:5060
qualify_frequency=60
[pbx_ata-identify]
type=identify
endpoint=pbx_ata
match=192.168.1.24/32
